cmake_minimum_required(VERSION 3.27)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(renderer_task_execution LANGUAGES CXX)
    include(CTest)
endif()
find_package(Threads REQUIRED)
add_library(renderer_task_execution INTERFACE)
add_library(renderer::task_execution ALIAS renderer_task_execution)
target_compile_features(renderer_task_execution INTERFACE cxx_std_20)
target_include_directories(renderer_task_execution INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(renderer_task_execution INTERFACE Threads::Threads)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
